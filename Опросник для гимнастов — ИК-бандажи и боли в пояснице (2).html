Как подключить вечную бесплатную базу на GitHub Pages за 1 минуту:
  1. Создайте пустую Google Таблицу (sheets.new) и нажмите в верхнем меню: Расширения → Apps Script.
  2. Замените весь текст в редакторе на готовый серверный код ниже и нажмите кнопку «Скопировать код скрипта»:
function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var list = [];
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][1]) {
      try { list.push(JSON.parse(rows[i][1])); } catch(e) {}
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', data: list }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) sheet.appendRow(['Дата', 'JSON_DATA']);
  var body = JSON.parse(e.postData.contents);
  if (body.action === 'clear') {
    sheet.clear();
    sheet.appendRow(['Дата', 'JSON_DATA']);
  } else if (body.action === 'addBatch' && Array.isArray(body.items)) {
    body.items.forEach(function(item) {
      sheet.appendRow([new Date(), JSON.stringify(item)]);
    });
  } else if (body.item) {
    sheet.appendRow([new Date(), JSON.stringify(body.item)]);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}

  2. Выберите тип «Веб-приложение», в поле «У кого есть доступ» обязательно выберите «Все» (Anyone) и нажмите «Начать развертывание».
